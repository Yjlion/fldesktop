find_package(Threads REQUIRED)
add_library(fl-files-search STATIC search.cpp)
target_include_directories(fl-files-search PUBLIC ${CMAKE_CURRENT_SOURCE_DIR})
target_link_libraries(fl-files-search PUBLIC Threads::Threads)
add_library(fl-files-rename STATIC rename_util.cpp)
target_include_directories(fl-files-rename PUBLIC ${CMAKE_CURRENT_SOURCE_DIR})
add_executable(fl-files main.cpp)
target_link_libraries(fl-files PRIVATE fl-files-search fl-files-rename fld Threads::Threads)
install(TARGETS fl-files RUNTIME DESTINATION ${CMAKE_INSTALL_BINDIR})
# removable drives through UDisks2 (optional)
if(TARGET fld-udisks)
  target_link_libraries(fl-files PRIVATE fld-udisks)
  target_compile_definitions(fl-files PRIVATE FLD_HAVE_UDISKS=1)
endif()
# network shares, phones and cameras through GVfs (optional)
if(TARGET fld-gvfs)
  target_sources(fl-files PRIVATE gvfs_ui.cpp)
  target_link_libraries(fl-files PRIVATE fld-gvfs)
  target_compile_definitions(fl-files PRIVATE FLD_HAVE_GVFS=1)
  if(TARGET fld-mdns) # shares that announce themselves on the network
    target_link_libraries(fl-files PRIVATE fld-mdns)
    target_compile_definitions(fl-files PRIVATE FLD_HAVE_MDNS=1)
  endif()
endif()
