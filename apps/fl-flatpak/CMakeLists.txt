pkg_check_modules(FLATPAK IMPORTED_TARGET flatpak)
pkg_check_modules(APPSTREAM IMPORTED_TARGET appstream>=1.0)
# the permission model is plain keyfile logic (unit tested without libflatpak)
add_library(fld-fp-perms STATIC permissions.cpp remotes.cpp)
target_include_directories(fld-fp-perms PUBLIC ${CMAKE_CURRENT_SOURCE_DIR})
target_link_libraries(fld-fp-perms PUBLIC fld)
if(FLATPAK_FOUND AND APPSTREAM_FOUND AND TARGET fld-glib)
  add_library(fld-fp STATIC fpclient.cpp)
  target_include_directories(fld-fp PUBLIC ${CMAKE_CURRENT_SOURCE_DIR})
  target_compile_options(fld-fp PRIVATE -Wno-cast-function-type)
  target_link_libraries(fld-fp PUBLIC fld-glib PkgConfig::FLATPAK PkgConfig::APPSTREAM)
  add_executable(fl-flatpak main.cpp prefs.cpp)
  target_link_libraries(fl-flatpak PRIVATE fld-fp fld-fp-perms)
  install(TARGETS fl-flatpak RUNTIME DESTINATION ${CMAKE_INSTALL_BINDIR})
  message(STATUS "libflatpak ${FLATPAK_VERSION}, appstream ${APPSTREAM_VERSION}: building fl-flatpak")
else()
  message(STATUS "libflatpak or appstream >= 1.0 not found: not building fl-flatpak")
endif()
