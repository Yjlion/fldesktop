# fl-fonts: preview, install and remove fonts (fontconfig)
add_library(fl-fonts-util STATIC fonts_util.cpp)
target_include_directories(fl-fonts-util PUBLIC ${CMAKE_CURRENT_SOURCE_DIR})
target_link_libraries(fl-fonts-util PUBLIC fld)
pkg_check_modules(FONTS_FC IMPORTED_TARGET fontconfig)
if(NOT FONTS_FC_FOUND)
  message(STATUS "fontconfig not found: fl-fonts is not built")
  return()
endif()
add_executable(fl-fonts main.cpp)
target_link_libraries(fl-fonts PRIVATE fl-fonts-util fld PkgConfig::FONTS_FC)
install(TARGETS fl-fonts RUNTIME DESTINATION ${CMAKE_INSTALL_BINDIR})
