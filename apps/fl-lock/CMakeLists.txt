find_package(Threads REQUIRED)
pkg_check_modules(LOCK_GIO REQUIRED IMPORTED_TARGET gio-2.0 gio-unix-2.0)
find_library(PAM_LIBRARY pam REQUIRED)
find_path(PAM_INCLUDE_DIR security/pam_appl.h REQUIRED)
add_executable(fl-lock main.cpp auth.cpp daemon.cpp)
target_include_directories(fl-lock PRIVATE ${PAM_INCLUDE_DIR})
target_link_libraries(fl-lock PRIVATE fld PkgConfig::LOCK_GIO ${PAM_LIBRARY} Threads::Threads)
# a finger unlocks too (fprintd)
if(TARGET fld-fprint)
  target_link_libraries(fl-lock PRIVATE fld-fprint)
  target_compile_definitions(fl-lock PRIVATE FLD_HAVE_FPRINT=1)
endif()
install(TARGETS fl-lock RUNTIME DESTINATION ${CMAKE_INSTALL_BINDIR})

# PAM service: the distributions name their common stack differently
if(EXISTS /etc/pam.d/system-auth OR FLD_PAM_STACK STREQUAL "system-auth")
  set(_pam_default "system-auth")
else()
  set(_pam_default "common-auth")
endif()
set(FLD_PAM_STACK "${_pam_default}" CACHE STRING "PAM stack fl-lock includes (system-auth on Arch, common-auth on Debian)")
if(FLD_PAM_STACK STREQUAL "common-auth")
  set(FLD_PAM_LINE "@include common-auth")
else()
  set(FLD_PAM_LINE "auth include ${FLD_PAM_STACK}")
endif()
configure_file(fl-lock.pam.in ${CMAKE_CURRENT_BINARY_DIR}/fl-lock.pam @ONLY)
install(FILES ${CMAKE_CURRENT_BINARY_DIR}/fl-lock.pam DESTINATION ${CMAKE_INSTALL_FULL_SYSCONFDIR}/pam.d RENAME fl-lock)
install(FILES ${CMAKE_SOURCE_DIR}/data/autostart/fl-lock-daemon.desktop
        DESTINATION ${CMAKE_INSTALL_FULL_SYSCONFDIR}/xdg/autostart)
