# fl-paint: a raster drawing program
add_library(fl-paint-util STATIC paint_util.cpp ora.cpp)
find_package(ZLIB REQUIRED)
target_link_libraries(fl-paint-util PUBLIC ZLIB::ZLIB)
target_include_directories(fl-paint-util PUBLIC ${CMAKE_CURRENT_SOURCE_DIR})
# pen pressure: the Wayland tablet protocol (stable in wayland-protocols 1.36+)
set(PAINT_PROTO)
set(_tablet_xml ${WAYLAND_PROTOCOLS_DIR}/stable/tablet/tablet-v2.xml)
set(_tablet_def)
if(NOT EXISTS ${_tablet_xml})
  set(_tablet_xml ${WAYLAND_PROTOCOLS_DIR}/unstable/tablet/tablet-unstable-v2.xml)
  set(_tablet_def FLD_TABLET_UNSTABLE=1)
endif()
fld_wayland_protocol(PAINT_PROTO ${_tablet_xml} client)
add_executable(fl-paint main.cpp pen.cpp ${PAINT_PROTO})
target_compile_definitions(fl-paint PRIVATE ${_tablet_def})
target_include_directories(fl-paint PRIVATE ${FLD_PROTO_OUT})
target_link_libraries(fl-paint PRIVATE fl-paint-util fld PkgConfig::WAYLAND_CLIENT)
install(TARGETS fl-paint RUNTIME DESTINATION ${CMAKE_INSTALL_BINDIR})
