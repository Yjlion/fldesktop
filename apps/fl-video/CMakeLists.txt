pkg_check_modules(MPV REQUIRED IMPORTED_TARGET mpv)
pkg_check_modules(VIDEO_EGL REQUIRED IMPORTED_TARGET egl)
set(VIDEO_PROTO)
fld_wayland_protocol(VIDEO_PROTO ${WAYLAND_PROTOCOLS_DIR}/unstable/idle-inhibit/idle-inhibit-unstable-v1.xml client)
add_library(fl-video-util STATIC video_util.cpp)
target_include_directories(fl-video-util PUBLIC ${CMAKE_CURRENT_SOURCE_DIR})
add_executable(fl-video main.cpp mpv_view.cpp idle_inhibit.cpp ${VIDEO_PROTO})
target_include_directories(fl-video PRIVATE ${FLD_PROTO_OUT})
# the OpenGL part of FLTK
if(FLTK_CONFIG)
  execute_process(COMMAND ${FLTK_CONFIG} --use-gl --use-images --ldflags OUTPUT_VARIABLE _vgl OUTPUT_STRIP_TRAILING_WHITESPACE)
  separate_arguments(_vgl UNIX_COMMAND "${_vgl}")
  target_link_libraries(fl-video PRIVATE ${_vgl})
else()
  target_link_libraries(fl-video PRIVATE fltk::gl)
endif()
find_package(OpenGL REQUIRED)
target_link_libraries(fl-video PRIVATE fl-video-util fld PkgConfig::MPV PkgConfig::VIDEO_EGL OpenGL::GL
                      PkgConfig::WAYLAND_CLIENT ${CMAKE_DL_LIBS})
install(TARGETS fl-video RUNTIME DESTINATION ${CMAKE_INSTALL_BINDIR})
