add_library(fl-view-util STATIC view_util.cpp)
target_include_directories(fl-view-util PUBLIC ${CMAKE_CURRENT_SOURCE_DIR})
add_executable(fl-view main.cpp)
target_link_libraries(fl-view PRIVATE fl-view-util fld)
# saving JPEG (PNG comes with FLTK)
find_package(JPEG QUIET)
if(JPEG_FOUND)
  target_link_libraries(fl-view PRIVATE JPEG::JPEG)
  target_compile_definitions(fl-view PRIVATE FLD_HAVE_JPEG=1)
endif()
install(TARGETS fl-view RUNTIME DESTINATION ${CMAKE_INSTALL_BINDIR})
