# GStreamer video in FLTK: used by fl-video and fl-webcam
pkg_check_modules(GST IMPORTED_TARGET gstreamer-1.0>=1.20 gstreamer-video-1.0 gstreamer-app-1.0 gstreamer-pbutils-1.0 gstreamer-tag-1.0 gstreamer-audio-1.0)
if(GST_FOUND)
  find_package(OpenGL REQUIRED)
  add_library(fld-gst STATIC frame_view.cpp)
  target_include_directories(fld-gst PUBLIC ${CMAKE_CURRENT_SOURCE_DIR})
  target_link_libraries(fld-gst PUBLIC fld PkgConfig::GST OpenGL::GL)
  # the OpenGL part of FLTK
  if(FLTK_CONFIG)
    execute_process(COMMAND ${FLTK_CONFIG} --use-gl --use-images --ldflags OUTPUT_VARIABLE _ggl OUTPUT_STRIP_TRAILING_WHITESPACE)
    separate_arguments(_ggl UNIX_COMMAND "${_ggl}")
    target_link_libraries(fld-gst PUBLIC ${_ggl})
  else()
    target_link_libraries(fld-gst PUBLIC fltk::gl)
  endif()
  message(STATUS "GStreamer ${GST_gstreamer-1.0_VERSION}: building fld-gst")
else()
  message(STATUS "GStreamer (gstreamer-1.0 and its base plugin libraries) not found: no fl-video and no fl-webcam")
endif()
