# libcups: pkg-config where available (Debian), else the library (Arch)
pkg_check_modules(CUPS IMPORTED_TARGET cups)
if(CUPS_FOUND)
  set(FLD_CUPS_LIB PkgConfig::CUPS)
else()
  find_path(CUPS_INCLUDE_DIR cups/cups.h)
  find_library(CUPS_LIBRARY cups)
  if(CUPS_INCLUDE_DIR AND CUPS_LIBRARY)
    add_library(fld_cups_imported INTERFACE)
    target_include_directories(fld_cups_imported INTERFACE ${CUPS_INCLUDE_DIR})
    target_link_libraries(fld_cups_imported INTERFACE ${CUPS_LIBRARY})
    set(FLD_CUPS_LIB fld_cups_imported)
    set(CUPS_FOUND ON)
  endif()
endif()

if(CUPS_FOUND)
  add_library(fld-printers STATIC cupsclient.cpp)
  target_include_directories(fld-printers PUBLIC ${CMAKE_CURRENT_SOURCE_DIR})
  # the CUPS 2.x API is used on purpose (CUPS 3 / libcups3 is a separate library)
  target_compile_options(fld-printers PRIVATE -Wno-deprecated-declarations)
  target_link_libraries(fld-printers PUBLIC ${FLD_CUPS_LIB})
  add_library(fld-printers-ui STATIC printers_ui.cpp)
  target_link_libraries(fld-printers-ui PUBLIC fld-printers fld)
  if(TARGET fld-mdns) # printers that announce themselves (IPP Everywhere) at once
    target_link_libraries(fld-printers-ui PUBLIC fld-mdns)
    target_compile_definitions(fld-printers-ui PRIVATE FLD_HAVE_MDNS=1)
  endif()
  message(STATUS "libcups found: building printer support")
else()
  message(STATUS "libcups not found: no printer support")
endif()
