set(_wlr_found FALSE)
foreach(v 0.20 0.19 0.18)
  pkg_check_modules(WLROOTS QUIET IMPORTED_TARGET wlroots-${v})
  if(WLROOTS_FOUND)
    message(STATUS "fl-wm: using wlroots ${WLROOTS_VERSION}")
    set(_wlr_found TRUE)
    break()
  endif()
endforeach()
if(NOT _wlr_found)
  message(FATAL_ERROR "fl-wm needs wlroots 0.18, 0.19 or 0.20 (or pass -DFLD_BUILD_WM=OFF)")
endif()

pkg_check_modules(WM_DEPS REQUIRED IMPORTED_TARGET
  wayland-server xkbcommon libinput pangocairo cairo libdrm pixman-1)

if(EXISTS ${WAYLAND_PROTOCOLS_DIR}/stable/tablet/tablet-v2.xml)
  fld_wayland_protocol(WM_PROTO_SRC ${WAYLAND_PROTOCOLS_DIR}/stable/tablet/tablet-v2.xml server)
endif()
fld_wayland_protocol(WM_PROTO_SRC ${CMAKE_SOURCE_DIR}/protocols/wlr-layer-shell-unstable-v1.xml server)
fld_wayland_protocol(WM_PROTO_SRC ${CMAKE_SOURCE_DIR}/protocols/fldesktop-window-v1.xml server)

# X11 programs through Xwayland, when wlroots was built with it
option(FLD_XWAYLAND "Run X11 programs in fl-wm through Xwayland (if wlroots supports it)" ON)
set(_wm_xwayland FALSE)
if(FLD_XWAYLAND)
  include(CheckCSourceCompiles)
  set(CMAKE_REQUIRED_INCLUDES ${WLROOTS_INCLUDE_DIRS})
  check_c_source_compiles("#include <wlr/config.h>
#if !WLR_HAS_XWAYLAND
#error no xwayland
#endif
int main(void) { return 0; }" FLWM_WLR_HAS_XWAYLAND)
  unset(CMAKE_REQUIRED_INCLUDES)
  pkg_check_modules(WM_XCB QUIET IMPORTED_TARGET xcb xcb-icccm)
  if(FLWM_WLR_HAS_XWAYLAND AND WM_XCB_FOUND)
    set(_wm_xwayland TRUE)
  endif()
endif()
if(_wm_xwayland)
  message(STATUS "fl-wm: X11 programs run through Xwayland")
  target_sources(fl-wm PRIVATE xwayland.c)
  target_compile_definitions(fl-wm PRIVATE FLWM_XWAYLAND)
  target_link_libraries(fl-wm PRIVATE PkgConfig::WM_XCB)
elseif(FLD_XWAYLAND)
  message(STATUS "fl-wm: wlroots without Xwayland (or no xcb-icccm), X11 programs will not run")
endif()
# wlroots headers include "<name>-protocol.h" for some protocols
file(GLOB _hdrs ${FLD_PROTO_OUT}/*-server-protocol.h)
foreach(h ${_hdrs})
  get_filename_component(n ${h} NAME)
  string(REPLACE "-server-protocol.h" "-protocol.h" alias ${n})
  configure_file(${h} ${FLD_PROTO_OUT}/server/${alias} COPYONLY)
endforeach()
target_include_directories(fl-wm PRIVATE ${FLD_PROTO_OUT}/server ${FLD_PROTO_OUT} ${CMAKE_CURRENT_SOURCE_DIR})
target_compile_definitions(fl-wm PRIVATE WLR_USE_UNSTABLE _GNU_SOURCE
                           FLD_PORTAL_EXEC="${CMAKE_INSTALL_FULL_LIBEXECDIR}/xdg-desktop-portal-fl")
target_link_libraries(fl-wm PRIVATE PkgConfig::WLROOTS PkgConfig::WM_DEPS m)
# SVG application icons in the Alt+Tab switcher (PNG icons work without it)
pkg_check_modules(WM_RSVG IMPORTED_TARGET librsvg-2.0>=2.46)
if(WM_RSVG_FOUND)
  target_compile_definitions(fl-wm PRIVATE HAVE_RSVG)
  target_link_libraries(fl-wm PRIVATE PkgConfig::WM_RSVG)
else()
  message(STATUS "fl-wm: librsvg not found, the window switcher shows only PNG icons")
endif()
install(TARGETS fl-wm RUNTIME DESTINATION ${CMAKE_INSTALL_BINDIR})
